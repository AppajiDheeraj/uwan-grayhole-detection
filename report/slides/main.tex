% ---------------- Introduction ----------------
\begin{frame}{Introduction \& Problem Statement}
  \begin{itemize}
    \item \textbf{Context:} Underwater Acoustic Sensor Networks (UASNs) carry sensor data to a surface sink over several acoustic hops (Network layer).
    \item \textbf{Problem:} A \textbf{grayhole} relay silently drops part of the packets it should forward; its loss blends with the normal loss of the channel.
    \item \textbf{Motivation:} The classic watchdog cannot tell a grayhole from an honest relay on a bad link, so it \textbf{accuses honest nodes} and routing around them costs delivery.
    \item \textbf{Objectives:}
      \begin{itemize}
        \item Judge forwarding only when the monitor could actually observe it.
        \item Compare observed forwards with what an honest relay \emph{should} show on each link.
        \item Evaluate against a fixed watchdog and a CAD-style detector in UnetStack.
      \end{itemize}
  \end{itemize}
\end{frame}

% ---------------- Background: setting ----------------
\begin{frame}{Background: Grayhole Attack}
  \begin{columns}[c]
    \column{0.52\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item Every relay must forward its neighbours' packets to the \textbf{sink}.
        \item A \textbf{blackhole} drops everything and is easy to find.
        \item A \textbf{grayhole} drops only a fraction (here \textbf{30\%}), so it looks like a lossy link.
        \item The \textbf{busiest relays} are made malicious: one grayhole can hurt many sensors.
      \end{itemize}
    \column{0.45\textwidth}
      \centering
      \includegraphics[height=0.68\textheight]{topology.pdf}\\
      {\scriptsize Test network 201: sink (green), relays (yellow), grayholes (red)}
  \end{columns}
\end{frame}

% ---------------- Background: why the watchdog fails ----------------
\begin{frame}{Background: Why the Watchdog Fails Under Water}
  \textbf{Watchdog} (Marti \emph{et al.} 2000; used in UnetStack3 by Chandavarkar \& Suresh 2021):
  send a packet, listen for the next hop's forward, accuse if too many are missing.

  \vspace{0.8em}
  \begin{block}{A forward can be missing because \dots}
    \begin{columns}[t]
      \column{0.24\textwidth}\centering{\color{bad}\Large\bfseries 1}\\\small the relay \textbf{dropped} it\\{\scriptsize\color{gray}(the attack)}
      \column{0.24\textwidth}\centering{\Large\bfseries 2}\\\small the monitor was \textbf{transmitting}\\{\scriptsize\color{gray}(half-duplex)}
      \column{0.24\textwidth}\centering{\Large\bfseries 3}\\\small the packet \textbf{collided} at the relay
      \column{0.24\textwidth}\centering{\Large\bfseries 4}\\\small the \textbf{long link} lost it
    \end{columns}
    \vspace{0.3em}
  \end{block}

  \vspace{0.5em}
  \centering
  Only cause 1 is an attack. Causes 2--4 \textbf{grow with traffic}, so a fixed threshold\\
  catches grayholes only by \textcolor{bad}{\textbf{also accusing honest relays}}.
\end{frame}

\begin{frame}{Proposed Approach: 1. Judge Only What Could Be Observed}
  \begin{columns}[c]
    \column{0.56\textwidth}
      \centering
      \begin{tikzpicture}[x=0.8cm,y=0.9cm,font=\scriptsize]
        \foreach \y/\n in {2/A,1/B,0/M} {\draw[-{Stealth[length=3pt]},gray] (0,\y)--(8,\y) node[right]{$t$}; \node[left] at (0,\y) {\small$\n$};}
        \fill[nitkblue!25,draw=nitkblue] (0.2,1.88) rectangle (1.2,2.12); \node[above] at (0.7,2.12) {$P$ sent};
        \fill[nitkblue!12,draw=nitkblue] (1.6,0.88) rectangle (2.6,1.12);
        \draw[densely dashed,gray] (0.2,2)--(1.6,1); \draw[densely dashed,gray] (1.2,2)--(2.6,1);
        \draw[decorate,decoration={brace,amplitude=3pt,mirror}] (1.6,0.8)--(2.6,0.8) node[midway,below=2pt]{$I_B$};
        \fill[good!30,draw=good] (3.0,0.88) rectangle (4.0,1.12); \node[above] at (3.5,1.12) {forward};
        \draw[densely dashed,gray] (3.0,1)--(3.7,0); \draw[densely dashed,gray] (4.0,1)--(4.7,0);
        \fill[good!15,draw=good] (3.7,-0.12) rectangle (4.7,0.12);
        \draw[bad,thick] (3.3,-0.35)--(3.3,0.35); \draw[bad,thick] (6.9,-0.35)--(6.9,0.35);
        \draw[bad,<->] (3.3,-0.5)--node[below]{watch window $W$}(6.9,-0.5);
      \end{tikzpicture}

      {\scriptsize From positions and the speed of sound (1500 m/s), monitor $M$ knows when packet $P$ reached relay $B$ ($I_B$) and when $B$'s forward should reach $M$ ($W$).}
    \column{0.42\textwidth}
      \begin{block}{A window counts only if}
        \begin{itemize}
          \item[\textbf{C1}] $M$ was silent during $W$
          \item[\textbf{C3}] $B$ was not sending during $I_B$
          \item[\textbf{C4}] no known frame hit $B$ during $I_B$
        \end{itemize}
      \end{block}
      \small Otherwise the window is \textbf{ignored}, not counted as a miss.
  \end{columns}
\end{frame}

% ---------------- Idea 2 ----------------
\begin{frame}{Proposed Approach: 2. Trust Against the Expected Loss}
  \begin{columns}[c]
    \column{0.52\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item Channel loss grows with distance: $e(d)=0.15\,(d/1500)^2$.
        \item Each kept window adds the chance an honest forward is heard: $g=(1-e(d_{AB}))(1-e(d_{BM}))$.
        \item Extra loss (collisions) is learned per \textbf{distance area} (near / medium / far) from the \textbf{other} relays' links: $\kappa_{\text{area}}$.
      \end{itemize}
    \column{0.45\textwidth}
      \begin{block}{Trust of relay $B$}
        \centering
        $E = G\,(1-\kappa_{\text{area}})$ \quad {\scriptsize expected forwards}\\[0.6em]
        $\text{Trust}_B = \dfrac{\sum \text{heard}}{\sum E}$\\[0.6em]
        $\text{Estimated drop} = 1-\text{Trust}_B$
      \end{block}
      \small A long or busy link is \emph{expected} to lose more, so it is not mistaken for a grayhole.
  \end{columns}
\end{frame}

% ---------------- Idea 3 ----------------
\begin{frame}{Proposed Approach: 3. Decide Carefully}
  \begin{columns}[t]
    \column{0.32\textwidth}
      \begin{block}{Dynamic threshold}
        \small Flag only if $\geq 20$ windows, drop $\geq 10\%$ and $z > z^*$; $z^*$ adapts to the number of relays and this round's spread.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Two-round confirmation}
        \small A relay becomes a \textbf{suspect} only if flagged in \textbf{two 30-min rounds in a row}.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Baseline cleaning}
        \small Suspects and relays with $\geq 10\%$ drop leave the ``normal loss'', so several grayholes cannot hide each other.
      \end{block}
  \end{columns}

  \vspace{1em}
  \begin{alertblock}{Response}
    The sink gives the administrator a \textbf{ranked suspect list} and routes traffic \textbf{around} suspects wherever another path exists.
  \end{alertblock}
\end{frame}

% ---------------- Implementation ----------------
\begin{frame}[fragile]{Implementation in UnetStack 7}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{itemize}\setlength\itemsep{0.4em}
        \item \texttt{MonitorNode}: one agent per node -- traffic, grayhole drops, watch windows, reports.
        \item \texttt{Detectors}: fixed watchdog, CAD-style and ours.
        \item \texttt{Topology}: deployment and routes that avoid suspects.
        \item \texttt{DistanceLossChannel}: loss grows with distance.
        \item One pipeline \texttt{run.sh}: record $\rightarrow$ compare $\rightarrow$ live network $\rightarrow$ graphs.
      \end{itemize}
    \column{0.5\textwidth}
\begin{lstlisting}
onPacket(pkt) {                 // the attack
  if (dropProb > 0 && random() < dropProb)
    return                      // grayhole drops it
  transmit(pkt)                 // forward
}
onRound(t) {                    // at the sink, every 30 min
  rows    = receivedEvidence()  // delivered reports
  flagged = peel(rows, ALPHA, suspects).flagged
  suspects += flagged.intersect(lastFlagged)
  lastFlagged = flagged
  nextHop = routesAvoiding(suspects)
}
\end{lstlisting}
  \end{columns}
\end{frame}

% ---------------- Discussion ----------------
\begin{frame}{Discussion}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{What each part contributes}
        \small
        \begin{itemize}
          \item No baseline cleaning: 3-grayhole detection falls 23/30 $\rightarrow$ 17/30.
          \item No observability filter: similar detection, slightly more false accusations (1.5\% vs 1.1\%).
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{alertblock}{Where we are not clearly ahead}
        \small
        \begin{itemize}
          \item Delivery gain over CAD-style is within the uncertainty.
          \item With 3 grayholes, CAD-style detects as often (77\%).
          \item Hotspot and 40-node networks: $\approx$10\% honest relays accused.
        \end{itemize}
      \end{alertblock}
  \end{columns}
\end{frame}

% ---------------- Conclusion ----------------
\begin{frame}{Conclusion \& Future Work}
  \begin{block}{Conclusion}
    \begin{itemize}
      \item Under water, a missing forward is often \textbf{not} an attack.
      \item Observable windows + expected link loss + dynamic threshold + two-round confirmation: \textbf{95\%} of grayholes caught, \textbf{0.3 vs 7.2} honest relays accused, highest delivery.
      \item Reproducible in UnetStack: \texttt{./run.sh demo} (3 min), \texttt{./run.sh} (all results).
    \end{itemize}
  \end{block}
  \begin{block}{Future Work}
    \begin{itemize}
      \item Dishonest monitors, mobile nodes and location errors.
      \item Learn the loss curve online; larger and denser networks.
    \end{itemize}
  \end{block}
\end{frame}
